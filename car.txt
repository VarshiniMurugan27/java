import java.io.*;

class Car {

    String brand;
    int year;

    Car(String b, int y) {
        brand = b;
        year = y;
    }

    void display() {
        System.out.println("Brand: " + brand);
        System.out.println("Year: " + year);
    }
}

public class ConstructorDemo {

    public static void main(String[] args) {

        Car car1 = new Car("Toyota", 2020);
        Car car2 = new Car("Honda", 2022);

        car1.display();
        System.out.println();

        car2.display();
    }
}
