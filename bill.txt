    Button b1, b2;

    public static void main(String args[]) {

        Bill f = new Bill();

        f.setTitle("STORE BILL");
        f.setSize(1000, 1000);
        f.setLayout(null);
        f.setVisible(true);
    }

    Bill() {

        l0 = new Label("VARSHINI SUPER MARKET");

        l9 = new Label("Total Amount");

        t1 = new TextField(30);
        t2 = new TextField(30);
        t3 = new TextField(30);
        t4 = new TextField(30);
        t5 = new TextField(30);

    public void actionPerformed(ActionEvent ae) {

        if (ae.getSource() == b1) {

            String s3 = t3.getText();
            String s4 = t4.getText();

            int a = Integer.parseInt(s3);
            int b = Integer.parseInt(s4);

            int tot = a * b;
            int gst = tot * 12 / 100;
            int amt = tot + gst;

            t5.setText("" + tot);
            t6.setText("" + gst);
            t7.setText("" + amt);
        }

        if (ae.getSource() == b2) {

            t1.setText("");
            t2.setText("");
            t3.setText("");
            t4.setText("");
            t5.setText("");
            t6.setText("");
            t7.setText("");
        }
    }
