import java.awt.*;
import java.awt.event.*;

public class Arith extends Frame implements ActionListener {

    Label l0, l1, l2, l3;
    TextField t1, t2, t3;
    Button b1, b2, b3, b4, b5;

    public static void main(String args[]) {

        Arith f = new Arith();

        f.setTitle("Arithmetic");
        f.setSize(500, 700);
        f.setLayout(null);
        f.setVisible(true);
    }

    Arith() {

        l0 = new Label("Arithmetic Operations");

        Font fo = new Font("Arial", Font.BOLD, 28);
        l0.setFont(fo);
        l0.setForeground(Color.RED);

        l1 = new Label("Number 1");
        l2 = new Label("Number 2");
        l3 = new Label("Result");

        t1 = new TextField(30);
        t2 = new TextField(30);
        t3 = new TextField(30);

        b1 = new Button("Add");
        b2 = new Button("Sub");
        b3 = new Button("Mul");
        b4 = new Button("Div");
        b5 = new Button("Clear");

        add(l0);
        l0.setBounds(100, 50, 300, 30);

        add(l1);
        l1.setBounds(50, 100, 80, 30);

        add(t1);
        t1.setBounds(150, 100, 100, 30);

        add(l2);
        l2.setBounds(50, 140, 80, 30);

        add(t2);
        t2.setBounds(150, 140, 100, 30);

        add(l3);
        l3.setBounds(50, 180, 80, 30);

        add(t3);
        t3.setBounds(150, 180, 100, 30);

        add(b1);
        b1.setBounds(50, 220, 50, 30);
        b1.addActionListener(this);

        add(b2);
        b2.setBounds(110, 220, 50, 30);
        b2.addActionListener(this);

        add(b3);
        b3.setBounds(170, 220, 50, 30);
        b3.addActionListener(this);

        add(b4);
        b4.setBounds(230, 220, 50, 30);
        b4.addActionListener(this);

        add(b5);
        b5.setBounds(290, 220, 50, 30);
        b5.addActionListener(this);

        // Close window
        addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent we) {
                System.exit(0);
            }
        });
    }

    public void actionPerformed(ActionEvent ae) {

        String s1 = t1.getText();
        String s2 = t2.getText();

        int a = Integer.parseInt(s1);
        int b = Integer.parseInt(s2);

        if (ae.getSource() == b1) {

            int c = a + b;
            t3.setText("" + c);
        }

        if (ae.getSource() == b2) {

            int c = a - b;
            t3.setText("" + c);
        }

        if (ae.getSource() == b3) {

            int c = a * b;
            t3.setText("" + c);
        }

        if (ae.getSource() == b4) {

            int c = a / b;
            t3.setText("" + c);
        }

        if (ae.getSource() == b5) {

            t1.setText("");
            t2.setText("");
            t3.setText("");
        }
    }
}
