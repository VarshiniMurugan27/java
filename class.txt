import java.io.*;

class Bsc {

    void java() {
        System.out.println("Java Program");
    }
}

class Aiml extends Bsc {

    void os() {
        System.out.println("Operating System");
    }
}

class Bca extends Aiml {

    void cws() {
        System.out.println("Cloud Web Service");
    }
}

class Multi {

    public static void main(String args[]) {

        Bsc a = new Bsc();
        a.java();

        Aiml b = new Aiml();
        b.os();
        b.java();

        Bca c = new Bca();
        c.cws();
        c.java();
        c.os();
    }
}
