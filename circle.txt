import java.awt.*;
import java.awt.event.*;

public class Circle extends Frame implements ActionListener {

    Label l0, l1, l2;
    TextField t1, t2;
    Button b1, b2, b3;

    public static void main(String args[]) {

        Circle f = new Circle();

        f.setTitle("Circle Program");
        f.setSize(500, 300);
        f.setLayout(null);
        f.setVisible(true);
    }

    Circle() {

        l0 = new Label("Circle Program");
        l1 = new Label("Radius");
        l2 = new Label("Result");

        Font fo = new Font("Arial", Font.BOLD, 24);
        l0.setFont(fo);

        t1 = new TextField(30);
        t2 = new TextField(30);

        b1 = new Button("Area");
        b2 = new Button("Circumference");
        b3 = new Button("Clear");

        l0.setBounds(150, 50, 200, 30);

        l1.setBounds(50, 100, 80, 30);

        l2.setBounds(50, 150, 80, 30);

        t1.setBounds(150, 100, 100, 30);

        t2.setBounds(150, 150, 100, 30);

        b1.setBounds(300, 100, 100, 30);

        b2.setBounds(300, 150, 100, 30);

        b3.setBounds(180, 210, 80, 30);

        add(l0);
        add(l1);
        add(l2);

        add(t1);
        add(t2);

        add(b1);
        add(b2);
        add(b3);

        b1.addActionListener(this);
        b2.addActionListener(this);
        b3.addActionListener(this);

        // Window close
        addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent we) {
                System.exit(0);
            }
        });
    }

    public void actionPerformed(ActionEvent ae) {

        if (ae.getSource() == b1) {

            String s1 = t1.getText();

            int r = Integer.parseInt(s1);

            double area = 3.14 * r * r;

            t2.setText("" + area);
        }

        if (ae.getSource() == b2) {

            String s2 = t1.getText();

            int r = Integer.parseInt(s2);

            double sur = 2 * 3.14 * r;

            t2.setText("" + sur);
        }

        if (ae.getSource() == b3) {

            t1.setText("");
            t2.setText("");
        }
    }
}
